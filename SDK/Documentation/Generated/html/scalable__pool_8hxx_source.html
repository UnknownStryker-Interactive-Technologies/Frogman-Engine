125 void _FE_VECTOR_CALL_ add_to_the_free_list(const block_info& block_p) noexcept
126 {
127 FE_NEGATIVE_ASSERT(m_free_list_size == possible_address_count, "Assertion Failure: The free list is full.");
128 block_info* const l_position = static_cast<block_info*>(m_free_list) + m_free_list_size;
129
130 l_position->_address = block_p._address;
131 l_position->_size_in_bytes = block_p._size_in_bytes;
132
133 ++(m_free_list_size);
134
135 if (m_is_page_heapified == true)
136 {
137 std::push_heap(static_cast<free_list_iterator>(m_free_list), static_cast<free_list_iterator>(m_free_list) + m_free_list_size, internal::pool::less_than{});

141 // Time complexity: O(2 log n)
142 FE::boolean _FE_VECTOR_CALL_ retrieve_from_the_free_list(internal::pool::block_info& out_alloc_result_p, FE::uint32 requested_bytes_p) noexcept
143 {
144 FE_ASSERT((requested_bytes_p % Alignment::size) == 0, "Critical Error in FE.Core.scalable_allocator: the requested allocation size '${%lu@0}' is not properly aligned by ${%lu@1}.", &requested_bytes_p, &Alignment::size);
145
146 if (m_is_page_heapified == false)
147 {
148 return _FE_FAILED_;
149 }
150
151 if (m_free_list_size == 0)
152 {
153 out_alloc_result_p._address = nullptr;
154 out_alloc_result_p._size_in_bytes = out_alloc_result_p._size_in_bytes xor out_alloc_result_p._size_in_bytes;
155 return _FE_FAILED_;
156 }
157
158 FE_ASSERT(m_is_page_heapified == true, "Assertion Failure: The page is not binary heapified.");
159 std::pop_heap(static_cast<free_list_iterator>(m_free_list), static_cast<free_list_iterator>(m_free_list) + m_free_list_size, internal::pool::less_than{});
160 --(m_free_list_size);
161
162 // Try allocation.
163 if (m_free_list[m_free_list_size]._size_in_bytes >= requested_bytes_p)
164 {
165 out_alloc_result_p._address = m_free_list[m_free_list_size]._address;
166 out_alloc_result_p._size_in_bytes = requested_bytes_p;
167 FE_ASSERT((reinterpret_cast<FE::uintptr>(out_alloc_result_p._address) % Alignment::size) == 0, "FE.Core.scalable_allocator has failed to allocate an address: the pointer value '${%p@0}' is not properly aligned by ${%lu@1}.", out_alloc_result_p._address, &Alignment::size);
168
169 m_free_list[m_free_list_size]._address += requested_bytes_p;
170 m_free_list[m_free_list_size]._size_in_bytes -= requested_bytes_p;
171
172 if (m_free_list[m_free_list_size]._size_in_bytes > 0)
173 {
174 add_to_the_free_list(m_free_list[m_free_list_size]);
175 }
176 return _FE_SUCCEEDED_;
177 }
178 else if (m_free_list[m_free_list_size]._size_in_bytes > 0) // Failed to find a block that fits the requested size.
179 {
180 add_to_the_free_list(m_free_list[m_free_list_size]);
181 }
182
183 out_alloc_result_p._address = nullptr;
184 out_alloc_result_p._size_in_bytes = out_alloc_result_p._size_in_bytes xor out_alloc_result_p._size_in_bytes;
185 return _FE_FAILED_; // Try iterate to the next page.
186 }

191 _FE_FORCE_INLINE_ PageListClass __select_page_list_by_capacity(FE::uint32 allocation_request_in_bytes_p) noexcept
192 {
193 FE_ASSERT(allocation_request_in_bytes_p <= FE::system_page_size);
194 constexpr FE::uint32 l_divider = 256; // 256 bytes
195 FE::uint32 l_quotient = allocation_request_in_bytes_p / l_divider;
196 /*
197 *
198 * l_quotient == 0 if 0~255 bytes are requested, fits within the pages with 6.25% capacity remaining.
199 * l_quotient == 1 if 256~511 bytes are requested
200 * l_quotient == 2 if 512~767 bytes are requested,
201 * l_quotient == 3 if 768~1023 bytes are requested
202 * l_quotient == 4 if 1024~1279 bytes are requested,
203 * l_quotient == 5 if 1280~1535 bytes are requested
204 * l_quotient == 6 if 1536~1791 bytes are requested
205 * l_quotient == 7 if 1792~2047 bytes are requested
206 * l_quotient == 8 if 2048~2303 bytes are requested,
207 * l_quotient == 9 if 2304~2559 bytes are requested
208 * l_quotient == 10 if 2560~2815 bytes are requested
209 * l_quotient == 11 if 2816~3071 bytes are requested
210 * l_quotient == 12 if 3072~3327 bytes are requested
211 * l_quotient == 13 if 3328~3583 bytes are requested
212 * l_quotient == 14 if 3584~3839 bytes are requested
213 * l_quotient == 15 if 3840~4095 bytes are requested
214 * l_quotient == 16 if 4096 bytes are requested,
215 */
216
217 if (allocation_request_in_bytes_p == 0)
218 {
219 PageListClass::_Unavailable;
220 }
221
222 switch (l_quotient)
223 {
224 case 0:
225 return PageListClass::_6_25_PercentRemaining;
226
227
228 case 1:
229 return PageListClass::_12_5_Percent;
230
231
232 case 2:
233 _FE_FALLTHROUGH_;
234 case 3:
235 return PageListClass::_25_Percent;
236
237
238 case 4:
239 _FE_FALLTHROUGH_;
240 case 5:
241 _FE_FALLTHROUGH_;
242 case 6:
243 _FE_FALLTHROUGH_;
244 case 7:
245 return PageListClass::_50_Percent;
246
247
248 case 8:
249 _FE_FALLTHROUGH_;
250 case 9:
251 _FE_FALLTHROUGH_;
252 case 10:
253 _FE_FALLTHROUGH_;
254 case 11:
255 return PageListClass::_75_Percent;
256
257
258 case 12:
259 _FE_FALLTHROUGH_;
260 case 13:
261 _FE_FALLTHROUGH_;
262 case 14:
263 _FE_FALLTHROUGH_;
264 case 15:
265 _FE_FALLTHROUGH_;
266 case 16:
267 return PageListClass::_100_Percent;
268
269 _FE_NODEFAULT_;
270 }

372 U* _FE_VECTOR_CALL_ allocate(FE::size size_p) noexcept
373 {
374 static_assert(std::is_array_v<U> == false, "Static Assertion Failed: The T must not be an array[] type.");
375 FE::uint32 l_queried_allocation_in_bytes = (FE::uint32)FE::calculate_aligned_memory_size_in_bytes<U, Alignment>(size_p);
376 FE_ASSERT((l_queried_allocation_in_bytes % Alignment::size) == 0, "Critical Error in FE.Core.scalable_allocator: the requested allocation size '${%d@0}' is not properly aligned by ${%lu@1}.", &l_queried_allocation_in_bytes, &Alignment::size);
377 FE_ASSERT(l_queried_allocation_in_bytes <= page_capacity, "Fatal Error: Unable to allocate ${%d0} bytes of memory that exceeds the pool chunk's capacity.", &size_p);
378 FE_ASSERT(m_pages_with_100_capacity.end() == nullptr, "Assertion failed: FE::list::end must return an iterator equivalent to a null pointer.");

385 {

388 if (l_page_list_iterator == m_pages_with_6_25_capacity.end())

392 }
393 break;

432 l_page_list_iterator = m_pages_with_100_capacity.begin();
433 if (l_page_list_iterator == m_pages_with_100_capacity.end())
434 {
435 m_pages_with_100_capacity.emplace_front();
436#if defined(_DEBUG_) || defined(_RELWITHDEBINFO_)
437 m_page_validation_table.insert(&m_pages_with_100_capacity.front());
438#endif
439 l_page_list_iterator = m_pages_with_100_capacity.begin();
440 }
441 break;
442
443 case internal::pool::PageListClass::_Unavailable:
444 return nullptr;
445
446 _FE_NODEFAULT_;
447 }
448
449 if (l_page_list_iterator->_remaining_capacity_in_bytes < l_queried_allocation_in_bytes)
450 {
451 if (l_required_page_class != internal::pool::PageListClass::_100_Percent)
452 {
453 l_required_page_class = static_cast<internal::pool::PageListClass>((int)l_required_page_class + 1);
454 l_page_list_iterator = m_pages_with_100_capacity.end();
455 continue;
456 }
457
458 m_pages_with_100_capacity.emplace_front();
459#if defined(_DEBUG_) || defined(_RELWITHDEBINFO_)
460 m_page_validation_table.insert(&m_pages_with_100_capacity.front());
461#endif
462 l_page_list_iterator = m_pages_with_100_capacity.begin();
463 break;
464 }
465 }
466 while (l_page_list_iterator == m_pages_with_100_capacity.end());
467
468
469 alignas(16) internal::pool::block_info l_memblock_info{};
470 var::boolean l_was_allocation_successful = __try_allocation_from_page(l_page_list_iterator, l_memblock_info, l_queried_allocation_in_bytes);
471 while (l_was_allocation_successful == false)
472 {
473 ++l_page_list_iterator;
474 if (l_page_list_iterator == m_pages_with_100_capacity.end())
475 {
476 m_pages_with_100_capacity.emplace_front();
477#if defined(_DEBUG_) || defined(_RELWITHDEBINFO_)
478 m_page_validation_table.insert(&m_pages_with_100_capacity.front());
479#endif
480 l_page_list_iterator = m_pages_with_100_capacity.begin();
481 }
482 l_was_allocation_successful = __try_allocation_from_page(l_page_list_iterator, l_memblock_info, l_queried_allocation_in_bytes);
483 }
484
485
486#if defined(_DEBUG_) || defined(_RELWITHDEBINFO_)
487 l_page_list_iterator->check_double_allocation(l_memblock_info);
488#endif
489 if constexpr (FE::is_trivial_v<U> == false)
490 {

504 switch (l_previous_class) // class before allocation
505 {
506 case internal::pool::PageListClass::_6_25_PercentRemaining:
507 l_previous_page_list = &m_pages_with_6_25_capacity;
508 break;

511 l_previous_page_list = &m_pages_with_12_5_capacity;
512 break;

514 case internal::pool::PageListClass::_25_Percent:
515 l_previous_page_list = &m_pages_with_25_capacity;
516 break;
517
518 case internal::pool::PageListClass::_50_Percent:
519 l_previous_page_list = &m_pages_with_50_capacity;
520 break;
521
522 case internal::pool::PageListClass::_75_Percent:
523 l_previous_page_list = &m_pages_with_75_capacity;
524 break;
525
526 case internal::pool::PageListClass::_100_Percent:
527 l_previous_page_list = &m_pages_with_100_capacity;
528 break;

531 }
532 FE_ASSERT(l_previous_page_list != nullptr, "Assertion Failure: The previous page list cannot be null.");
533
534 l_page_list_iterator->_remaining_capacity_in_bytes -= l_memblock_info._size_in_bytes;
535
536 internal::pool::PageListClass l_new_class = internal::pool::__select_page_list_by_capacity(l_page_list_iterator->_remaining_capacity_in_bytes);
537 if (l_new_class != l_previous_class) // if the page has changed its class after allocation, move it to the corresponding list.
538 {
539 switch (l_new_class) // new class after allocation
540 {
541 case internal::pool::PageListClass::_Unavailable:
542 m_unavailable_pages.splice( FE::iterator_cast<typename pool_type::const_iterator>(m_unavailable_pages.rbegin()), *l_previous_page_list, FE::iterator_cast<typename pool_type::const_iterator>(l_page_list_iterator));
543 break;

715private:
716 /* Time complexity:
717 Allocation from a stack: O(1)
718 Allocation from a heap tree: O(2 log n)
719 Defragmentation: O(5n + n log n) [rarely gets called]
720
721 Best: O(1)
722 Worst: O(5n + n log n) + O(2 log n)
723 */
724 static FE::boolean _FE_VECTOR_CALL_ __try_allocation_from_page(page_iterator page_p, internal::pool::block_info& out_result_p, FE::uint32 bytes_p) noexcept
725 {
726 FE_ASSERT((bytes_p % Alignment::size) == 0, "Critical Error in FE.Core.scalable_allocator: the requested allocation size '${%lu@0}' is not properly aligned by ${%lu@1}.", &bytes_p, &Alignment::size);
727 if (page_p->is_page_heapified() == true)
728 {
729 if(page_p->retrieve_from_the_free_list(out_result_p, bytes_p) == _FE_FAILED_)
730 {
731 // Try defragmenting the page.
732 __defragment(page_p);
733 FE::boolean l_was_successful = page_p->retrieve_from_the_free_list(out_result_p, bytes_p); // Retry it. Traverse to the nxt page if it fails.
734 FE_ASSERT(out_result_p._size_in_bytes <= page_capacity, "Critical Error in FE.Core.scalable_allocator: value overflow suspected.");
735 return l_was_successful;
736 }
737 FE_ASSERT(out_result_p._size_in_bytes <= page_capacity, "Critical Error in FE.Core.scalable_allocator: value overflow suspected.");
738 return _FE_SUCCEEDED_;
739 }
740 else
741 {
742 // Try allocation by pushing the stack pointer.
743 out_result_p._address = page_p->_page_iterator;
744 out_result_p._size_in_bytes = bytes_p;
745 page_p->_page_iterator += bytes_p;
746
747 // The requested allocation size overflows the page capacity.
748 if (page_p->_page_iterator > page_p->get_page_end())
749 {
750 page_p->_page_iterator -= bytes_p; // Allocation failed, unwind the stack to cancel the allocation.
751 out_result_p._address = nullptr;
752 out_result_p._size_in_bytes = out_result_p._size_in_bytes xor out_result_p._size_in_bytes;
753
754 if (page_p->get_free_list_size() > 1) // Can the free list defragmented?
755 {
756 __defragment(page_p); // Defragment the page.
757 }
758
759 if ((page_p->get_free_list_size() > 0) &&
760 (page_p->is_page_heapified() == true)) // Is the free list not empty?
761 {
762 //FE_ASSERT(page_p->is_page_heapified() == true, "Assertion Failure: The page must be heapified after defragmentation.");
763
764 FE::boolean l_was_successful = page_p->retrieve_from_the_free_list(out_result_p, bytes_p); // Try allocating from the defragmented free list. Traverse to the next page if it fails.
765 FE_ASSERT(out_result_p._size_in_bytes <= page_capacity, "Critical Error in FE.Core.scalable_allocator: value overflow suspected.");
766 return l_was_successful;
767 }
768 // If the page is out of capacity and the free list is empty, the pool needs to create a new page.
769 return _FE_FAILED_;
770 }
771 }
772 FE_ASSERT(out_result_p._size_in_bytes <= page_capacity, "Critical Error in FE.Core.scalable_allocator: value overflow suspected.");
773 return _FE_SUCCEEDED_;
774 }
775
776 // Try reducing the pointer offset data to uint32 to save memory, since the page size is no larger than 2MiB.
777 static void _FE_VECTOR_CALL_ __defragment(page_iterator page_p) noexcept
778 {
779 if (page_p->get_free_list_size() <= 1) _FE_UNLIKELY_
780 {
781 return;
782 }
783 std::sort<free_list_iterator, internal::pool::from_low_address>(page_p->get_free_list(),
784 page_p->get_free_list() + page_p->get_free_list_size(),
785 internal::pool::from_low_address{}
786 );
787 // Merge the free list.
788 free_list_iterator l_iterator = page_p->get_free_list();
789 free_list_iterator l_next = l_iterator + 1;
790 free_list_iterator l_end = l_iterator + page_p->get_free_list_size();
791
792 // Time complexity: O(n)
793 while (l_next != l_end)
794 {
795 // Merge the adjacent blocks.
796 if ((l_iterator->_address + l_iterator->_size_in_bytes) == l_next->_address)
797 {
798 l_iterator->_size_in_bytes += l_next->_size_in_bytes;
799
800 // Nullify the block.
801 l_next->_address = nullptr;
802 l_next->_size_in_bytes = l_next->_size_in_bytes xor l_next->_size_in_bytes;
803 ++l_next; // Look for the next block.
804 continue;
805 }
806 // Move to the next block if they are not adjacent.
807 l_iterator = l_next;
808 ++l_next;
809 }
810
811 // Migrate null blocks to right-side of the array to exclude them from being binary searched.
812 /* Time complexity:
813 Best - O(n/2)
814 Worst - O(n)
815 */
816 l_end = std::partition(page_p->get_free_list(), l_end,
817 [](const internal::pool::block_info& block_p)
818 {
819 return block_p._address != nullptr;
820 }
821 );
822 // Reset it.
823 page_p->set_free_list_size( static_cast<FE::uint32>( l_end - page_p->get_free_list() ) );
824
825 // Heapify the free list. Time complexity: O(3n)
826 std::make_heap(page_p->get_free_list(), l_end, internal::pool::less_than{});
827
828 page_p->set_page_heapified(); // Switch the allocation strategy to binary search.
829 }

1002 _FE_FORCE_INLINE_ FE::internal::pool::PageListClass __select_page_list_by_capacity(FE::uint32 allocation_request_in_bytes_p) noexcept
1003 {
1004 FE_ASSERT(allocation_request_in_bytes_p <= FE::system_large_page_size);
1005 constexpr FE::uint32 l_divider = 131072; // 131072 bytes because 4096*512 == 2MiB and 256*512 == 131072.
1006 FE::uint32 l_quotient = allocation_request_in_bytes_p / l_divider;
1007 /*
1008 * The exact byte numbers below change if _LARGE_PAGES_ is defined as 1.
1009 *
1010 * l_quotient == 0 if 0~255 bytes are requested, fits within the pages with 6.25% capacity remaining.
1011 * l_quotient == 1 if 256~511 bytes are requested
1012 * l_quotient == 2 if 512~767 bytes are requested,
1013 * l_quotient == 3 if 768~1023 bytes are requested
1014 * l_quotient == 4 if 1024~1279 bytes are requested,
1015 * l_quotient == 5 if 1280~1535 bytes are requested
1016 * l_quotient == 6 if 1536~1791 bytes are requested
1017 * l_quotient == 7 if 1792~2047 bytes are requested
1018 * l_quotient == 8 if 2048~2303 bytes are requested,
1019 * l_quotient == 9 if 2304~2559 bytes are requested
1020 * l_quotient == 10 if 2560~2815 bytes are requested
1021 * l_quotient == 11 if 2816~3071 bytes are requested
1022 * l_quotient == 12 if 3072~3327 bytes are requested
1023 * l_quotient == 13 if 3328~3583 bytes are requested
1024 * l_quotient == 14 if 3584~3839 bytes are requested
1025 * l_quotient == 15 if 3840~4095 bytes are requested
1026 * l_quotient == 16 if 4096 bytes are requested,
1027 */
1028
1029 if (allocation_request_in_bytes_p == 0)
1030 {

1537 Defragmentation: O(5n + n log n) [rarely gets called]
1538
1539 Best: O(1)
1540 Worst: O(5n + n log n) + O(2 log n)
1541 */
1542 static FE::boolean _FE_VECTOR_CALL_ __try_allocation_from_page(page_iterator page_p, internal::pool::block_info& out_result_p, FE::uint32 bytes_p) noexcept
1543 {
1544 FE_ASSERT((bytes_p % Alignment::size) == 0, "Critical Error in FE.Core.scalable_allocator: the requested allocation size '${%lu@0}' is not properly aligned by ${%lu@1}.", &bytes_p, &Alignment::size);
1545 if (page_p->is_page_heapified() == true)
1546 {
1547 if (page_p->retrieve_from_the_free_list(out_result_p, bytes_p) == _FE_FAILED_)
1548 {
1549 // Try defragmenting the page.
1550 __defragment(page_p);
1551 FE::boolean l_was_successful = page_p->retrieve_from_the_free_list(out_result_p, bytes_p); // Retry it. Traverse to the nxt page if it fails.
1552 FE_ASSERT(out_result_p._size_in_bytes <= page_capacity, "Critical Error in FE.Core.scalable_allocator: value overflow suspected.");
1553 return l_was_successful;
1554 }
1555 FE_ASSERT(out_result_p._size_in_bytes <= page_capacity, "Critical Error in FE.Core.scalable_allocator: value overflow suspected.");
1556 return _FE_SUCCEEDED_;
1557 }
1558 else
1559 {
1560 // Try allocation by pushing the stack pointer.
1561 out_result_p._address = page_p->_page_iterator;
1562 out_result_p._size_in_bytes = bytes_p;
1563 page_p->_page_iterator += bytes_p;
1564
1565 // The requested allocation size overflows the page capacity.
1566 if (page_p->_page_iterator > page_p->get_page_end())
1567 {
1568 page_p->_page_iterator -= bytes_p; // Allocation failed, unwind the stack to cancel the allocation.
1569 out_result_p._address = nullptr;
1570 out_result_p._size_in_bytes = out_result_p._size_in_bytes xor out_result_p._size_in_bytes;
1571
1572 if (page_p->get_free_list_size() > 1) // Can the free list defragmented?
1573 {
1574 __defragment(page_p); // Defragment the page.
1575 }
1576
1577 if ((page_p->get_free_list_size() > 0) &&
1578 (page_p->is_page_heapified() == true)) // Is the free list not empty?
1579 {
1580 //FE_ASSERT(page_p->is_page_heapified() == true, "Assertion Failure: The page must be heapified after defragmentation.");
1581
1582 FE::boolean l_was_successful = page_p->retrieve_from_the_free_list(out_result_p, bytes_p); // Try allocating from the defragmented free list. Traverse to the next page if it fails.
1583 FE_ASSERT(out_result_p._size_in_bytes <= page_capacity, "Critical Error in FE.Core.scalable_allocator: value overflow suspected.");
1584 return l_was_successful;
1585 }
1586 // If the page is out of capacity and the free list is empty, the pool needs to create a new page.
1587 return _FE_FAILED_;
1588 }
1589 }
1590 FE_ASSERT(out_result_p._size_in_bytes <= page_capacity, "Critical Error in FE.Core.scalable_allocator: value overflow suspected.");
1591 return _FE_SUCCEEDED_;
1592 }

1594 // Try reducing the pointer offset data to uint32 to save memory, since the page size is no larger than 2MiB.
1595 static void _FE_VECTOR_CALL_ __defragment(page_iterator page_p) noexcept
1596 {
1597 if (page_p->get_free_list_size() <= 1) _FE_UNLIKELY_
1598 {
1599 return;
1600 }
1601 std::sort<std::execution::parallel_unsequenced_policy, free_list_iterator, internal::pool::from_low_address>( std::execution::parallel_unsequenced_policy{},
1602 page_p->get_free_list(),
1603 page_p->get_free_list() + page_p->get_free_list_size(),
1604 internal::pool::from_low_address{});
1605 // Merge the free list.
1606 free_list_iterator l_iterator = page_p->get_free_list();
1607 free_list_iterator l_next = l_iterator + 1;
1608 free_list_iterator l_end = l_iterator + page_p->get_free_list_size();
1609
1610 // Time complexity: O(n)
1611 while (l_next != l_end)
1612 {
1613 // Merge the adjacent blocks.
1614 if ((l_iterator->_address + l_iterator->_size_in_bytes) == l_next->_address)
1615 {
1616 l_iterator->_size_in_bytes += l_next->_size_in_bytes;
1617
1618 // Nullify the block.
1619 l_next->_address = nullptr;
1620 l_next->_size_in_bytes = l_next->_size_in_bytes xor l_next->_size_in_bytes;
1621 ++l_next; // Look for the next block.
1622 continue;
1623 }
1624 // Move to the next block if they are not adjacent.
1625 l_iterator = l_next;
1626 ++l_next;
1627 }
1628
1629 // Migrate null blocks to right-side of the array to exclude them from being binary searched.
1630 /* Time complexity:
1631 Best - O(n/2)
1632 Worst - O(n)
1633 */
1634 l_end = std::partition(std::execution::par_unseq, page_p->get_free_list(), l_end,
1635 [](const internal::pool::block_info& block_p)
1636 {
1637 return block_p._address != nullptr;
1638 }
1639 );
1640 // Reset it.
1641 page_p->set_free_list_size(static_cast<FE::uint32>(l_end - page_p->get_free_list()));
1642
1643 // Heapify the free list. Time complexity: O(3n)
1644 std::make_heap(page_p->get_free_list(), l_end, internal::pool::less_than{});
1645
1646 page_p->set_page_heapified(); // Switch the allocation strategy to binary search.
1647 }
1648 };
