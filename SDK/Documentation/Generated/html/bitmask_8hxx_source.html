147 constexpr bool read_at(FE::size index_p) const noexcept
148 {
149 static_assert( sizeof(m_64bit_buffer) == 8 );
150 FE_ASSERT(index_p < m_capacity_in_bits, "Assertion failed: index is out of bounds of the bitmask capacity.");
151
152 if (__large_bitset_engaged())
153 {
154 FE::size l_slot_index = index_p / 8;
155 FE::size l_bit_index = index_p % 8;
156
157 var::byte l_8bit_slot = m_bitmask[l_slot_index] << (7 - l_bit_index);
158 l_8bit_slot = l_8bit_slot >> 7;
159 return static_cast<bool>(l_8bit_slot);
160 }
161 else
162 {
163 var::uint64 l_64bit_slot = m_64bit_buffer << (63 - index_p);
164 l_64bit_slot = l_64bit_slot >> 63;
165 return static_cast<bool>(l_64bit_slot);
166 }
167 }

169 constexpr void write_at(FE::size index_p, bool value_p) noexcept
170 {
171 static_assert(sizeof(m_64bit_buffer) == 8);
172 FE_ASSERT(index_p < m_capacity_in_bits, "Assertion failed: index is out of bounds of the bitmask capacity.");
173
174 if (__large_bitset_engaged())
175 {
176 FE::size l_slot_index = index_p / 8;
177 FE::byte l_bit_index = static_cast<FE::byte>(index_p % 8);
178
179 var::byte l_8bit_slot = 0b00000001;
180 l_8bit_slot = l_8bit_slot << l_bit_index;
181 if (value_p == true)
182 {
183 m_bitmask[l_slot_index] = m_bitmask[l_slot_index] | l_8bit_slot;
184 }
185 else
186 {
187 m_bitmask[l_slot_index] = m_bitmask[l_slot_index] ^ l_8bit_slot;
188 }
189 }
190 else
191 {
192 var::uint64 l_64bit_slot = 0b00000000'00000000'00000000'00000000'00000000'00000000'00000000'00000001;
193 l_64bit_slot = l_64bit_slot << index_p;
194 if (value_p == true)
195 {
196 m_64bit_buffer = m_64bit_buffer | l_64bit_slot;
197 }
198 else
199 {
200 m_64bit_buffer = m_64bit_buffer ^ l_64bit_slot;
201 }
202 }

206 constexpr void reserve(FE::size new_capacity_in_bits_p) noexcept
207 {
208 FE::size l_new_size_in_bytes = __calculate_size_of_bits_in_bytes(new_capacity_in_bits_p);
209 FE::size l_current_size_in_bytes = __calculate_size_of_bits_in_bytes(m_capacity_in_bits);
210
211 if (l_current_size_in_bytes >= l_new_size_in_bytes)
212 {
213 return;
214 }
215
216 var::byte* l_new_bitmask = m_allocator.allocate(l_new_size_in_bytes);
217 std::memset(l_new_bitmask, 0, l_new_size_in_bytes);
218
219 if (m_bitmask != nullptr)
220 {
221 FE::memcpy(l_new_bitmask, l_new_size_in_bytes,
222 m_bitmask, l_current_size_in_bytes);
223 m_allocator.deallocate(m_bitmask, l_current_size_in_bytes);
224 }
225 else
226 {
227 *((var::uint64*)l_new_bitmask) = m_64bit_buffer;
228 }
229 m_bitmask = l_new_bitmask;
230 m_capacity_in_bits = new_capacity_in_bits_p;
231 }
