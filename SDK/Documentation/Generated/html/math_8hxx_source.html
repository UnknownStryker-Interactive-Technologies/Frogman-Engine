168 return lhs_p._x < rhs_p._x;

175 template <Interpolation Mode = Interpolation::_Linear>
176 constexpr FE::float64 f(FE::float64 x_p) noexcept
177 {
178 const point2D l_value =
179 {
180 ._x = x_p,
181 ._y = 0.0
182 };
183
184 const auto l_s_comparator = [](const point2D& lhs_p, const point2D& rhs_p) noexcept
185 {
186 return lhs_p._x < rhs_p._x;
187 };
188
189 const auto l_next = std::lower_bound(m_lut.begin(), m_lut.end(), l_value, l_s_comparator);
190 FE_ASSERT(l_next != m_lut.end());
191
192 if (l_next->_x == x_p)
193 {
194 return l_next->_y;
195 }
196
197 const auto l_prev = std::prev(l_next);
198 FE_ASSERT(l_prev != m_lut.end());
199
200
201 if constexpr (Mode == Interpolation::_Linear)
202 {
203 /*
204 y = mx;
205
206 y/x = m;
207
208 For two points:
209 (y2 - y1) / (x2 - x1) = m.
210 */
211 FE::float64 l_delta_y = l_next->_y - l_prev->_y;
212 FE::float64 l_delta_x = l_next->_x - l_prev->_x;
213
214 FE_ASSERT(l_delta_x != 0.0);
215
216 FE::float64 l_slope = l_delta_y / l_delta_x;
217 /*
218 y = mx;
219
220 y_p - y1 = m(x_p - x1);
221 y_p = m(x_p - x1) + y1;
222 */
223 return l_slope * (x_p - l_prev->_x) + l_prev->_y;
224 }
225 else if constexpr (Mode == Interpolation::_Step)
226 {
227 return l_prev->_y;
228 }
229 else if constexpr (Mode == Interpolation::_SmoothStep)
230 {
231 }
232 else if constexpr (Mode == Interpolation::_CubicHermite)
233 {
234 }
235 else if constexpr (Mode == Interpolation::_MonotoneCubic)
236 {
237 }
238 }
239};
