36/*
37fqueue is a fixed-capacity, circular queue implementation that provides efficient FIFO operations.
38It uses a contiguous memory block to store elements and manages front and back pointers to simulate a queue.
39Supports both trivial and non-trivial types through the Traits template parameter.
40Capacity is fixed at compile-time, and operations are noexcept where possible.
41*/
42template<class T, size Capacity, class Traits = internal::memory_traits<T>>
43class fqueue final
44{
45 static_assert((std::is_same<T, typename Traits::value_type>::value), "Static Assertion Failed: The template argument T and Traits' value_type have be the same type.");
46 static_assert(std::is_class<Traits>::value, "Static Assertion Failed: The template argument Traits is not a class type.");
47
48public:
49 using value_type = T;
50 using size_type = var::uint64;
51 using reference = T&;
52 using const_reference = const T&;
53 using pointer = T*;
54 using const_pointer = const T*;
55 using const_iterator = FE::const_iterator<FE::contiguous_iterator<T>>;
56 using const_reverse_iterator = FE::const_reverse_iterator<FE::contiguous_iterator<T>>;

90 fqueue(InputIterator begin_p, InputIterator end_p) noexcept
91 : m_memory(),
92 m_front_ptr(reinterpret_cast<pointer>(m_memory)),

98 static_assert((std::is_same<typename std::remove_const<typename InputIterator::value_type>::type, typename std::remove_const<value_type>::type>::value), "Static Assertion Failure: InputIterator's value_type has to be the same as fqueue's value_type.");
99
100 FE_NEGATIVE_ASSERT(begin_p >= end_p, "${%s@0}: The input iterator ${%s@1} must not be greater than ${%s@2}.", TO_STRING(FE::ErrorCode::_FatalMemoryError_1XX_InvalidIterator), TO_STRING(begin_p), TO_STRING(end_p));
101 FE_NEGATIVE_ASSERT(static_cast<uint64>(end_p - begin_p) > Capacity, "${%s@0}: The input size exceeds the fqueue capacity.", TO_STRING(FE::ErrorCode::_FatalMemoryError_1XX_BufferOverflow));
102
103 Traits::copy_construct(InputIterator{ m_begin_ptr }, begin_p, end_p - begin_p);
104 }

106 fqueue(const fqueue& other_p) noexcept
107 : m_memory(),

178 template <typename... Arguments>
179 constexpr void push(Arguments&&... value_p) noexcept
180 {
181 if (m_back_ptr >= m_begin_ptr + Capacity)
182 {
183 __set_back_pointer_to_zero();
184 FE_NEGATIVE_ASSERT(m_back_ptr >= m_front_ptr, "${%s@0}: Exceeded the queue index boundary", TO_STRING(FE::ErrorCode::_FatalMemoryError_1XX_AccessViolation));
185 }
186
187 if constexpr (Traits::is_trivial == TypeTriviality::_NotTrivial)
188 {
189 new(m_back_ptr) T( std::forward<Arguments&&>(value_p)... );
190 }
191 else if constexpr (Traits::is_trivial == TypeTriviality::_Trivial)
192 {
193 *m_back_ptr = T(std::forward<Arguments&&>(value_p)...);
194 }
195
196 ++m_back_ptr;
197 ++m_size;
198 }
199
200 constexpr value_type pop() noexcept
201 {
202 if ((m_begin_ptr + Capacity) == m_front_ptr)
203 {
204 m_front_ptr = m_begin_ptr;
205 FE_NEGATIVE_ASSERT(is_empty() == true, "${%s@0}: Exceeded the queue index boundary", TO_STRING(FE::ErrorCode::_FatalMemoryError_1XX_AccessViolation));
206 }
207
208 T l_return_value_buffer = std::move(*m_front_ptr);
209
210 if constexpr (Traits::is_trivial == TypeTriviality::_NotTrivial)
211 {
212 m_front_ptr->~T();
213 }
214
215 ++m_front_ptr;
216 --m_size;
217 return l_return_value_buffer;
218 }
219
220 constexpr void pop_all() noexcept
221 {
222 if (is_empty() == false)
223 {
224 if constexpr (Traits::is_trivial == TypeTriviality::_NotTrivial)
225 {
226 if (m_back_ptr >= m_front_ptr)
227 {
228 Traits::destruct(m_front_ptr, m_back_ptr);
229 }
230 else
231 {
232 Traits::destruct(m_begin_ptr, m_back_ptr);
233 Traits::destruct(m_front_ptr, m_begin_ptr + Capacity);
234 }
235 }
236
237 __set_front_pointer_to_zero();
238 __set_back_pointer_to_zero();
239 m_size = 0;
240 }
