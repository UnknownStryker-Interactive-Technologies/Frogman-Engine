44template<typename T>
45_FE_FORCE_INLINE_ constexpr HashInputDataType evaluate_hash_input_data_type()
46{
47 if constexpr (FE::is_constant_string_v<T>)
48 {
49 return HashInputDataType::_CString;
50 }
51 else if constexpr (std::is_pointer_v<T>)
52 {
53 return HashInputDataType::_Address;
54 }
55 else if constexpr (FE::is_string_class_v<T>)
56 {
57 return HashInputDataType::_StringClass;
58 }
59 else if constexpr (std::same_as<T, FE::bitmask>)
60 {
61 return HashInputDataType::_Bitmask;
62 }
63
64 return HashInputDataType::_Binary;
65}

79/*
80The FE::hash class template provides a hashing mechanism for objects of type T
81allowing the use of different hashing algorithms (such as Robin Hood Hashing or CityHash) based on the specified HasherType
82and is specialized for handling address-based hashing with the HashInputDataType::_Address type.
83*/
84template<typename T, HasherType HasherType = HasherType::_CityHash, HashInputDataType HashInputDataType = FE::evaluate_hash_input_data_type<T>()>
85class hash;
