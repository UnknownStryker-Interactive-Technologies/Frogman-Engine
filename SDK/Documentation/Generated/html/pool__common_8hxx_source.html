132 _FE_FORCE_INLINE_ void __enable_large_pages() noexcept
133 {
134 HANDLE l_token;
135 if (!OpenProcessToken(GetCurrentProcess(), TOKEN_ADJUST_PRIVILEGES | TOKEN_QUERY, &l_token))
136 {
137 FE_EXIT_IF(true, FE::ErrorCode::_FatalWinAPI_Error_4XX_OpenProcessTokenFailure, "Frogman Engine Runtime Error: Failed to enable large pages.");
138 return;
139 }
140
141 TOKEN_PRIVILEGES l_tp;
142 LookupPrivilegeValue(NULL, SE_LOCK_MEMORY_NAME, &l_tp.Privileges[0].Luid);
143
144 l_tp.PrivilegeCount = 1;
145 l_tp.Privileges[0].Attributes = SE_PRIVILEGE_ENABLED;
146
147 AdjustTokenPrivileges(l_token, FALSE, &l_tp, 0, NULL, NULL);
148 FE_EXIT_IF(GetLastError() != 0, FE::ErrorCode::_FatalWinAPI_Error_4XX_AdjustTokenPrivilegesFailure, "Frogman Engine Runtime Error: Failed to enable large pages.");
149 CloseHandle(l_token);
150 }
