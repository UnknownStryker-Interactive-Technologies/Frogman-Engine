195 size_type try_push_back(const value_type& value_p) noexcept
196 {
197 boost::shared_lock_guard<SharedMutex> l_lock(m_length_modifier_lock);
198 size_type l_idx = m_size.fetch_add(1, std::memory_order_acq_rel);
199 size_type l_current_capacity = m_capacity.load(std::memory_order_acquire);
200 /*
201 I have spent days to figure out why the program arbitrarily crashes when try_reserve() is called concurrently.
202 m_size has to be less than or equal to m_capacity. However, m_size sometimes becomes m_size+1.😭
203 To explain the unpleasant situation:
204
205 Step one: a thread comes here and fails to try_push_back() and sets m_size to the current capacity
206 because the given index points outside of the array range [the array ranges from 0 to the current capacity]. Note that m_size is always equal to the given index + 1 at the moment of 'size_type l_idx = m_size.fetch_add(1, std::memory_order_acq_rel);'.
207
208 Step two: another thread enters try_push_back() and fails too, and post-increments one to m_size.
209
210 Step three: the first thread successfully acquries a chance to operate try_reserve(), and it reads the post-incremented value, thus m_size > m_capacity sometimes becomes true.
211
212 Step four: segfault.
213 */
214 if (l_idx >= l_current_capacity)
215 {
216 //std::cerr << m_size.load(std::memory_order_acquire) << '\n';
217 /*
218 Atomic writes are expansive than the reads, because only a single thread can write to a cachline at a time.
219 The performance characteristics of atmoic operations vary on the hardware architectures.
220 On some hardwares, they use some kind of bus to communicate and synchronize their data.
221 If mutiple threads reference the same variable on the cache, then it is true sharing.
222 If multple threads reference the same address range on the cache and do not share the data, then it is false sharing.
223 True sharing and false sharing are expansive; remember that avoiding resource sharing avoids synchronization costs.
224 */
225 m_size.store(l_current_capacity, std::memory_order_release);
226 return std::numeric_limits<size_type>::max();
227 }
228 //assert(l_idx != m_capacity);
229
230 // Commit the transaction.
231 new (m_active.load(std::memory_order_acquire) + l_idx) T(value_p);
232 return l_idx;
233 }

276 bool try_reserve(const size_type new_capacity_p) noexcept
277 {
278 pointer l_nullptr = nullptr;
279
280 if (new_capacity_p <= m_capacity.load(std::memory_order_acquire))
281 {
282 return true;
283 }
284
285 if (m_reserved.compare_exchange_strong(l_nullptr, reinterpret_cast<pointer>(this), std::memory_order_acq_rel) == true)
286 {
287 m_reserved.store(m_allocator.allocate(new_capacity_p), std::memory_order_release);
288 //std::cout << new_capacity_p << '\n';
289 pointer l_retired = nullptr;
290 size_type l_retired_size = 0;
291 size_type l_retired_capacity = 0;
292 { // temporarily blocks other threads from reading and writing the array.
293 std::lock_guard<SharedMutex> l_lock(m_length_modifier_lock);
294 l_retired = m_active.exchange(m_reserved.load(std::memory_order_relaxed), std::memory_order_relaxed);
295 l_retired_size = m_size.load(std::memory_order_relaxed);
296 l_retired_capacity = m_capacity.exchange(new_capacity_p, std::memory_order_relaxed);
297 __move_construct_from(l_retired, std::min(l_retired_size, l_retired_capacity) /* read the comment in the try_emplace_back() */);
298 }
299 __tear_down(l_retired, l_retired_size, l_retired_capacity);
300 m_reserved.store(nullptr, std::memory_order_release);
301 return true;
302 }
303 return false;
304 }

306 size_type push_back(const value_type& value_p) noexcept
307 {
308 size_type l_idx = try_push_back(value_p);
309 while (l_idx == std::numeric_limits<size_type>::max())
310 {
311 size_type l_new_capacity = m_capacity.load(std::memory_order_acquire);
312 l_new_capacity += (l_new_capacity >> 1);
313 try_reserve(l_new_capacity);
314 l_idx = try_push_back(value_p);
315 }
316 return l_idx;
317 }
