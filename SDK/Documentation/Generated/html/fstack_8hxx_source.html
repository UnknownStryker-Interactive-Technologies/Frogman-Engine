36The FE::fstack class template is a fixed-capacity stack implementation that utilizes custom memory traits for managing its elements
37providing various constructors and assignment operators for initialization and manipulation of the stack's contents.
38*/
39template<class T, size Capacity, class Traits = FE::internal::memory_traits<T>>
40class fstack final
41{
42 static_assert((std::is_same<T, typename Traits::value_type>::value), "Static Assertion Failed: The template argument T and Traits' value_type have be the same type.");
43 static_assert(std::is_class<Traits>::value, "Static Assertion Failed: The template argument Traits is not a class type.");
44
45public:
46 using value_type = T;
47 using length_type = var::size;
48 using size_type = var::size;
49 using reference = T&;
50 using const_reference = const T&;
51 using pointer = T*;
52 using const_pointer = const T*;
53 using const_iterator = FE::const_iterator<FE::contiguous_iterator<T>>;
54 using const_reverse_iterator = FE::const_reverse_iterator<FE::contiguous_iterator<T>>;

70 fstack(std::initializer_list<value_type>&& initializer_list_p) noexcept

81 template<class InputIterator>
82 fstack(InputIterator begin_p, InputIterator end_p) noexcept
83 : m_memory(),
84 m_top_ptr(reinterpret_cast<pointer>(m_memory) + (end_p - begin_p)),
85 m_begin_ptr(reinterpret_cast<pointer>(m_memory))
86 {
87 static_assert(std::is_class<InputIterator>::value, "Static Assertion Failure: The template argument InputIterator must be a class type.");
88 static_assert((std::is_same<typename std::remove_const<typename InputIterator::value_type>::type, typename std::remove_const<value_type>::type>::value), "Static Assertion Failure: InputIterator's value_type has to be the same as fstack's value_type.");
89
90 FE_NEGATIVE_ASSERT(begin_p >= end_p, "${%s@0}: The input iterator ${%s@1} must not be greater than the iterator ${%s@2}.", TO_STRING(FE::ErrorCode::_FatalMemoryError_1XX_InvalidIterator), TO_STRING(begin_p), TO_STRING(end_p));
91 FE_NEGATIVE_ASSERT(static_cast<uint64>(end_p - begin_p) > Capacity, "${%s@0}: The input size exceeds the fstack capacity.", TO_STRING(FE::ErrorCode::_FatalMemoryError_1XX_BufferOverflow));

93 Traits::copy_construct(InputIterator{ m_begin_ptr }, begin_p, end_p - begin_p);

96 fstack(fstack& other_p) noexcept

127 fstack& operator=(std::initializer_list<value_type> initializer_list_p) noexcept
128 {
129 FE_NEGATIVE_ASSERT(initializer_list_p.size() > Capacity, "ERROR!: The length of std::initializer_list exceeds the Capacity");

131
132 if (is_empty())

134 this->~fstack();

191 void push(const value_type& value_p) noexcept
192 {
193 FE_NEGATIVE_ASSERT(m_top_ptr >= m_begin_ptr + Capacity, "${%s@0}: The fstack top exceeded the index boundary", TO_STRING(ErrorCode::_FatalMemoryError_1XX_AccessViolation));
194
195 if constexpr (Traits::is_trivial == TypeTriviality::_NotTrivial)
196 {
197 new(m_top_ptr) T(value_p);
198 }
199 else if constexpr (Traits::is_trivial == TypeTriviality::_Trivial)
200 {
201 *m_top_ptr = value_p;
202 }
203
204 ++m_top_ptr;
205 }
206
207 value_type pop() noexcept
208 {
209 FE_NEGATIVE_ASSERT(is_empty() == true, "${%s@0}: The fstack top index reached zero. The index value_p must be greater than zero", TO_STRING(ErrorCode::_FatalMemoryError_1XX_AccessViolation));
210
211 --m_top_ptr;
212 T l_return_value_buffer = std::move(*m_top_ptr);
213
214 if constexpr (Traits::is_trivial == TypeTriviality::_NotTrivial)
215 {
216 m_top_ptr->~T();
217 }
218
219 return l_return_value_buffer;
220 }

222 void pop_all() noexcept
223 {
224 if (is_empty() == false)
225 {
226 if constexpr (Traits::is_trivial == TypeTriviality::_NotTrivial)
227 {
228 Traits::destruct(m_begin_ptr, m_top_ptr);
229 }
230 __set_top_pointer_to_zero();
231 }

315 void __restructure_fstack_with_move_semantics(value_type* const source_begin_p, FE::size source_size_p) noexcept
316 {
317 FE::size l_this_size = size();
318
319 if (source_size_p > l_this_size)
320 {
321 FE::size l_count_to_construct = source_size_p - l_this_size;
322 Traits::move_construct(m_top_ptr, source_begin_p - l_count_to_construct, l_count_to_construct);
323
324 Traits::move_assign(m_begin_ptr, source_begin_p, l_this_size);
325 }
326 else if (source_size_p < l_this_size)
327 {
328 FE::size l_count_to_destruct = l_this_size - source_size_p;
329
330 if constexpr (Traits::is_trivial == TypeTriviality::_NotTrivial)
331 {
332 Traits::destruct(m_top_ptr - l_count_to_destruct, m_top_ptr);
333 }
334
335 Traits::move_assign(m_begin_ptr, source_begin_p, source_size_p);
336 }
337 else
338 {
339 Traits::move_assign(m_begin_ptr, source_begin_p, source_size_p);
340 }
341 }
