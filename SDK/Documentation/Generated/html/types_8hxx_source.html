148template <typename T>
149class lazy_const
150{
151public:
152 using value_type = typename std::remove_const<T>::type;
153 using reference = value_type&;
154 using const_reference = const value_type&;
155
156private:
157 value_type m_data;
158 bool m_is_initialized;
159
160public:
161 constexpr lazy_const() noexcept : m_data(), m_is_initialized(false) {}
162 constexpr lazy_const(value_type&& data_p) noexcept : m_data(std::move(data_p)), m_is_initialized(true) {}
163 constexpr ~lazy_const() noexcept {};
164
165 constexpr lazy_const(const lazy_const& other_p) noexcept : m_data(other_p.m_data), m_is_initialized(true) {}
166 constexpr lazy_const(lazy_const&& rvalue_p) noexcept : m_data(std::move(rvalue_p.m_data)), m_is_initialized(true) {}
167
168 constexpr lazy_const& operator=(value_type&& data_p) noexcept
169 {
170 if (m_is_initialized == true)
171 {
172 assert(false && "Frogman Engine C++: lazy_const cannot be set more than once.");
173 std::exit(-9);
174 }
175
176 m_data = std::move(data_p);
177 m_is_initialized = true;
178
179 return *this;
180 }
181
182 constexpr lazy_const& operator=(const lazy_const& other_p) noexcept
183 {
184 if (m_is_initialized == true)
185 {
186 assert(false && "Frogman Engine C++: lazy_const cannot be set more than once.");
187 std::exit(-9);
188 }
189
190 m_data = other_p.m_data;
191 m_is_initialized = true;
192
193 return *this;
194 }
195
196 constexpr lazy_const& operator=(lazy_const&& rvalue_p) noexcept
197 {
198 if (m_is_initialized == true)
199 {
200 assert(false && "Frogman Engine C++: lazy_const cannot be set more than once.");
201 std::exit(-9);
202 }
203
204 m_data = std::move(rvalue_p.m_data);
205 m_is_initialized = true;
206
207 return *this;
208 }
209
210 _FE_FORCE_INLINE_ const_reference load() const noexcept { return m_data; }
211};
