60 The partition_stable function template sorts a range of elements defined by two iterators based on a specified predicate
61 stable partitioning of elements based on exclusion_target_p.
62 Groups all elements NOT equal to exclusion_target_p at one end (based on IsolationVector).
63 Preserves relative order of retained elements.
64 Useful when element order matters (e.g., rendering, logic sequencing).
65*/
66template<IsolationVector IsolationVector, class Iterator, class Predicate>
67constexpr FE::pair<Iterator, Iterator> partition_stable_if(Iterator begin_p, Iterator end_p, Predicate predicate_p)
68{
69 if constexpr (IsolationVector == IsolationVector::_Right)
70 {
71 Iterator l_end = end_p;
72 Iterator l_begin = begin_p;
73 Iterator l_tmp_it = begin_p;
74
75 while (l_tmp_it != l_end)
76 {
77 if (predicate_p(*l_begin))
78 {
79 if (!predicate_p(*l_tmp_it))
80 {
81 std::swap(*l_begin, *l_tmp_it);
82 ++l_begin;
83 ++l_tmp_it;
84 continue;
85 }
86
87 ++l_tmp_it;
88 continue;
89 }
90
91 ++l_begin;
92 ++l_tmp_it;
93 }
94 return FE::pair<Iterator, Iterator>{begin_p, l_tmp_it};
95 }
96 else if constexpr (IsolationVector == IsolationVector::_Left)
97 {
98 Iterator l_rend = begin_p;
99 Iterator l_rbegin = std::next(begin_p, ((end_p - begin_p) - 1));
100 Iterator l_rtmp_it = l_rbegin;
101
102 while (l_rtmp_it > l_rend)
103 {
104 if (predicate_p(*l_rbegin))
105 {
106 if (!predicate_p(*l_rtmp_it))
107 {
108 std::swap(*l_rbegin, *l_rtmp_it);
109 --l_rbegin;
110 --l_rtmp_it;
111 continue;
112 }
113
114 --l_rtmp_it;
115 continue;
116 }
117
118 --l_rbegin;
119 --l_rtmp_it;
120 }
121 std::swap(*l_rbegin, *l_rtmp_it);
122 if (predicate_p(*l_rbegin))
123 {
124 ++l_rbegin;
125 }
126 return FE::pair<Iterator, Iterator>{l_rbegin, end_p};
127 }
128}

203/*
204 - Time complexity -
205 Best: O(n/2)
206 Worst: O(n)
207
208 The partition_unstable function template sorts a range of elements defined by two iterators based on a specified predicate
209 partitioning the elements into two groups according to the provided IsolationVector.
210 Unstable in-place partitioning based on exclusion_target_p.
211 Swaps elements from both ends to isolate non-excluded elements.
212 Does NOT preserve relative order.
213 Optimized for performance in large datasets or non-order-sensitive contexts.
214*/
215template<IsolationVector IsolationVector, class Iterator, class Predicate>
216constexpr FE::pair<Iterator, Iterator> partition_unstable_if(Iterator begin_p, Iterator end_p, Predicate predicate_p)
217{
218 Iterator l_begin = begin_p;
219 Iterator l_end = end_p;
220 --end_p;
221 FE_NEGATIVE_ASSERT(begin_p >= end_p, "Assertion failure: the 'begin' iterator is pointing after the 'end' iterator.");
222
223
224 if constexpr (IsolationVector == IsolationVector::_Right)
225 {
226 while (begin_p < end_p)
227 {
228 if (predicate_p(*begin_p) && !predicate_p(*end_p))
229 {
230 std::swap(*begin_p, *end_p);
231 }
232
233 if (!predicate_p(*begin_p))
234 {
235 ++begin_p;
236 }
237
238 if (predicate_p(*end_p))
239 {
240 --end_p;
241 }
242 }
243 if (begin_p >= end_p)
244 {
245 end_p = begin_p;
246 }
247 FE_NEGATIVE_ASSERT(l_begin > end_p, "Assertion failure: the begin iterator is pointing after the end iterator.");
248 return FE::pair<Iterator, Iterator>{l_begin, end_p};
249 }
250 else if constexpr (IsolationVector == IsolationVector::_Left)
251 {
252 while (begin_p <= end_p)
253 {
254 if (!predicate_p(*begin_p) && predicate_p(*end_p))
255 {
256 std::swap(*begin_p, *end_p);
257 }
258
259 if (!predicate_p(*end_p))
260 {
261 --end_p;
262 }
263
264 if (predicate_p(*begin_p))
265 {
266 ++begin_p;
267 }
268 }
269 FE_NEGATIVE_ASSERT(begin_p > l_end, "Assertion failure: the begin iterator is pointing after the end iterator.");
270 return FE::pair<Iterator, Iterator>{begin_p, l_end};
271 }
