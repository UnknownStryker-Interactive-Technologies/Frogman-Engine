188// The FE::is_power_of_two function is a constexpr function that checks if a given size_t value is a power of two by using a bitwise operation.
189_FE_CONSTEXPR17_ FE::boolean is_power_of_two(FE::size value_p) noexcept
190{
191 if (value_p == 0)
192 {
193 return false;
194 }
195
196 /* Since 4 is a power of 2 and
197 0b0001 == 1
198 0b0010 == 2
199 0b0011 == 3
200 0b0100 == 4
201 ...
202
203 if you can see the pattern, you can find that
204 4 - 1 is 3 and equals to 0b0011 in binary.
205
206 0b0100
207 & 0b0011
208 --------
209 0b0000
210
211 Therefore,
212 0b0100 AND 0b0011 == 0b0000.
213 */
214 return (value_p & (value_p - 1)) == 0;
215}
