234/*
235Frogman Engine serialization & deserialization system:
236
237 Compatible component types: Plain old data, trivial fixed-sized arrays, FE smart pointers, std::basic_string, std::array, and std::vector.
238
239 * C/C++ object that might contain a chain of derivation, components, other supported types, and a component's chain of derivation.
240
241 * Raw pointers, non-trivial fixed-sized arrays, and references are not supported.
242*/
243/*
244memory layout reflection for serialization |--- etc...
245 *
246 |
247 Shallower <----- Memory Hierarchy Depth ----> Deeper Lower Memory Address
248 |----------------| |
249 | - FE::string - | |
250 |----------------------------------| ----------| FE::smart_ptr | |
251 | Target Entry Non-Trivial Object | | |length, capacity| |
252 |- member variables - | | |----------------| |
253 | FE::string m_raw_name ----------|----| |
254 | FE::vector<float64, 3> m_vector |-------| |
255 |----------------------------------| | |----------------| |
256 | | - FE::vector - | |
257 |-------| x | |
258 | y | |
259 | z | |
260 |----------------| Higher Memory Address
261 |
262 *
263
264Memory Layer Traversal Order: Entry.FE::string m_raw_name -> FE::string.FE::smart_ptr -> FE::smart_ptr.m_smart_ptr data
265 |
266 |-----------------------------------------------|
267 |
268 |--> FE::string.length -> FE::string.capacity
269 |
270 |
271 |---------------------------------------|
272 |
273 |--> FE::vector.x -> FE::vector.y -> FE::vector.z
274*/
275struct property_metadata
276{
277 FE::TypeTriviality _is_trivial;
278 var::boolean _is_serializable;
279 var::uint32 _size_in_byte;
280 var::ptrdiff _offset_from_this;
281
282 FE::ASCII* _name;
283 FE::ASCII* m_typename;
284};

377 void register_property(const C& host_class_instance_p, const T& property_p, const std::string_view& property_name_p) noexcept
378 {

380 static_assert((std::is_reference_v<T> == false) && (std::is_pointer_v<T> == false), "Static assertion failure: raw pointers and references cannot be serialized nor deserialized.");
381 static_assert(! ((FE::is_trivial<T>::value == false) && (std::is_array_v<T>)), "Static assertion failure: fixed-sized non-trivial arrays are not serializable nor deserializable.");
382 FE_ASSERT(property_name_p.empty() != true, "Assertion failure: property name cannot be null.");
383
384 property_metadata l_property_meta_data;
385 l_property_meta_data._is_trivial = static_cast<TypeTriviality>(FE::is_trivial<T>::value);
386 l_property_meta_data._is_serializable = FE::is_serializable<T>::value;
387 static_assert(sizeof(T) <= FE::uint32_max, "Static assertion failure: the property instance size is too enormous.");
388 l_property_meta_data._size_in_byte = sizeof(T);
389 l_property_meta_data._offset_from_this = (reinterpret_cast<FE::byte* const>(&property_p) - reinterpret_cast<FE::byte*>(&host_class_instance_p));
390 l_property_meta_data._name = property_name_p.data();
391 l_property_meta_data.m_typename = reflection::type_id<T>().name();
392
393 std::lock_guard<lock_type> l_lock(m_lock);
394 std::pmr::string l_host_class_instance_typename(reflection::type_id<C>().name(), m_pool);
395 auto l_iterator = m_property_registry.find(l_host_class_instance_typename);
396 if (FE_UNLIKELY(l_iterator == m_property_registry.end())) _FE_UNLIKELY_
397 {
398 auto l_result = m_property_registry.emplace(l_host_class_instance_typename, class_property_list(m_pool));

402
403

409 framework_base::get_framework().get_method_reflection().register_task< FE::cpp_style_task<property_registry, void(void*)>>(__get_deserialization_task_name(l_property_meta_data.m_typename), &property_registry::__deserialize_by_foreach_mutually_recursive<T>);
410
411 if constexpr (FE::has_value_type<T>::value == true)
412 {
413 if constexpr ((FE::is_serializable<typename T::value_type>::value == true) &&
414 (FE::is_trivial<typename T::value_type>::value == false))
415 {
416 __push_multidimensional_container_serialization_task_recursive<typename T::value_type>();
417 __push_multidimensional_container_deserialization_task_recursive<typename T::value_type>();
418 }
419 }
420 else if constexpr (FE::has_element_type<T>::value == true)
421 {
422 if constexpr ((FE::is_serializable<typename T::element_type>::value == true) &&

425 __push_multidimensional_container_serialization_task_recursive<typename T::element_type>();

428 }
429 }

434
435

446 "FE Runtime Reflection Registration Error: failed to associate a property metadata in the FE runtime reflection system lookup table.\nThis error might have ocurred in the FE runtime reflection metadata loader function, during the initialization.");

450 template<typename T>
451 void serialize(std::pmr::string& out_ret_buffer_p, const T& object_p, FE::ASCII* const version_p) noexcept
452 {
453 static_assert(std::is_class_v<T>, "Non-class/struct field variables cannot be serialized.");
454 static_assert((std::is_reference_v<T> == false) && (std::is_pointer_v<T> == false), "static assertion failure: raw pointers and references cannot be serialized nor deserialized.");
455 out_ret_buffer_p.clear();
456 out_ret_buffer_p.reserve(one_KiB); // Pre-allocate 1 KiB.
457
458 std::lock_guard<lock_type> l_lock(m_lock);
459 std::pmr::string l_typename(reflection::type_id<T>().name(), m_pool);
460 auto l_search_result = m_property_registry.find(l_typename);
461 if (l_search_result == m_property_registry.end())
462 {
463#ifdef _ENABLE_LOG_
464 constexpr FE::ASCII* l_error_code = TO_STRING(ErrorCode::_FatalSerializationError_3XX_TypeNotFound);
465 constexpr FE::ASCII* l_error_message = "serialization failed - could not find the requested type information";
466 FE_LOG(FE::log::Severity::_Warning, "Frogman Engine ${%s@0}: ${%s@1}.", l_error_code, l_error_message);
467#endif
468 return;
469 }
470 m_class_layer.emplace_back(&(l_search_result->second), l_search_result->second.begin());
471
472 if constexpr (FE::has_base_type<T>::value == true)
473 {
474 __push_parent_class_layers_recursive<T>();
475 }
476 out_ret_buffer_p += version_p;
477 out_ret_buffer_p += "\n";
478 out_ret_buffer_p += l_typename;
479 out_ret_buffer_p += "{";
480 __serialize_mutually_recursive<T>(out_ret_buffer_p, object_p);
481 out_ret_buffer_p += "};$-";
482
483 while (m_scalable_container_size_record.empty() == false)
484 {
485 std::pmr::string l_buffer(m_pool);
486 l_buffer.reserve(FE::algorithm::utility::count_int_digit_length(m_scalable_container_size_record.front()));
487 FE::algorithm::utility::uint_to_string(l_buffer.data(), l_buffer.capacity(), m_scalable_container_size_record.front());
488
489 out_ret_buffer_p += l_buffer.c_str();
490 out_ret_buffer_p += "-";
491 m_scalable_container_size_record.pop_front();
492 }
493 out_ret_buffer_p += "EOF\0";
494 }

497 template<typename T>
498 void deserialize(const std::pmr::string& data_p, T& out_object_p, _FE_MAYBE_UNUSED_ FE::ASCII* const version_p) noexcept
499 {
500 static_assert(std::is_class_v<T>, "Non-class/struct field variables cannot be serialized.");
501 static_assert((std::is_reference_v<T> == false) && (std::is_pointer_v<T> == false), "static assertion failure: raw pointers and references cannot be serialized nor deserialized.");
502
503 if (data_p.empty() == true)
504 {
505#ifdef _ENABLE_LOG_
506 constexpr FE::ASCII* l_error_code = TO_STRING(ErrorCode::_FatalDeserializationError_3XX_FileBufferEmpty);
507 constexpr FE::ASCII* l_error_message = "deserialization failed - the input data buffer is empty";
508 FE_LOG(FE::log::Severity::_Warning, "Frogman Engine ${%s@0}: ${%s@1}.", l_error_code, l_error_message);
509#endif
510 return;
511 }
512
513 std::lock_guard<lock_type> l_lock(m_lock);
514 std::pmr::string l_typename(reflection::type_id<T>().name(), m_pool);
515 auto l_search_result = m_property_registry.find(l_typename);
516 FE_EXIT_IF((l_search_result == m_property_registry.end()) || (l_search_result->second.size() == 0), ErrorCode::_FatalSerializationError_3XX_TypeNotFound, "serialization failed: could not find the requested type information or the class/struct is empty");
517 m_class_layer.emplace_back(&(l_search_result->second), l_search_result->second.begin());
518
519 if constexpr (FE::has_base_type<T>::value == true)
520 {
521 __push_parent_class_layers_recursive<T>();
522 }
523
524 m_input_buffer = data_p;
525 // Checks the file version
526 FE_EXIT_IF(!algorithm::string::compare_ranged<char>(m_input_buffer.data(), algorithm::string::range{ 0, std::strlen(version_p) },
527 version_p, algorithm::string::range{ 0, std::strlen(version_p) }),
528 FE::ErrorCode::_FatalSerializationError_3XX_FileVersionMismatch, "Assertion failure: the serialization file version is not supported.");
529
530 m_position = m_input_buffer.begin();
531 m_position += m_input_buffer.find('{');
532 FE_NEGATIVE_ASSERT(m_position == m_input_buffer.end(), "The serialization file is ill-formed or unsupported.");
533
534 // Checks the class type name
535 FE_EXIT_IF(!algorithm::string::compare_ranged<char>(l_typename.data(), algorithm::string::range{0, std::strlen(l_typename.data())},
536 m_input_buffer.c_str(), algorithm::string::range{ std::strlen(version_p) + sizeof('\n'), static_cast<uint64>(m_position - m_input_buffer.begin())}),
537 FE::ErrorCode::_FatalSerializationError_3XX_TypeMismatch, "Unable to deserialize an instance with a different class name.");
538 ++m_position; // Point the first byte.
539
540 auto l_size_indicator = m_input_buffer.begin();
541 auto l_pos = m_input_buffer.rfind("$-");
542 FE_ASSERT(l_pos != m_input_buffer.npos, "Assertion failure: the serialization file is ill-formed or unsupported.");
543
544 l_size_indicator += l_pos;
545 l_size_indicator += 2; // to skip the "$-" and point to the first one.
546
547 std::pmr::string::size_type l_eof_pos = m_input_buffer.find("EOF\0");
548 FE_ASSERT(l_eof_pos != m_input_buffer.npos, "Assertion failure: the serialization file is ill-formed or unsupported.");
549 auto l_file_end = m_input_buffer.begin() + l_eof_pos;
550 FE_ASSERT(l_file_end <= m_input_buffer.end(), "Assertion failure: the serialization file is ill-formed or unsupported.");
551 while (l_size_indicator != l_file_end)
552 {
553 algorithm::utility::uint_info l_info = algorithm::utility::string_to_uint(FE::iterator_cast<FE::ASCII*>(l_size_indicator));
554 FE_LOG_IF(l_info._value == 0, FE::log::Severity::_Warning, "Warning: the size of the container is zero. Please debug if the file is corrupted.");
555 m_scalable_container_size_record.push_back(l_info._value);
556 l_size_indicator += l_info._digit_length; // move to the next.
557 ++l_size_indicator; // to skip the '-'.
558 }
559
560 __deserialize_mutually_recursive<T>(out_object_p);
561 }

653 template<typename T>
654 void __serialize_mutually_recursive(std::pmr::string& out_ret_buffer_p, const T& object_p) noexcept
655 {
656 var::ptrdiff l_offset_from_the_upmost_base_class_instance = 0;
657 while (m_class_layer.empty() == false)
658 {
659 switch (__get_metadata_of_the_property(__get_the_top_class_property_list_iterator())._is_trivial)
660 {
661 case TypeTriviality::_Trivial:
662 {
663 // Check if the field variable meta data is valid.
664 FE_NEGATIVE_ASSERT(__get_metadata_of_the_property(__get_the_top_class_property_list_iterator())._size_in_byte == 0, "Assertion failed: unable to serialize a zero-byte property.");
665
666 out_ret_buffer_p.append(reinterpret_cast<const char*>(&object_p) + l_offset_from_the_upmost_base_class_instance + __get_memory_offset_of_the_property( __get_the_top_class_property_list_iterator() ),
667 __get_metadata_of_the_property( __get_the_top_class_property_list_iterator() )._size_in_byte);
668
669 // Look for the next registered property of the class.
670 ++(__get_the_top_class_property_list_iterator());
671
672 // Pop the class layer if the iterator reached the end of the property list.
673 if (__get_the_top_class_property_list_iterator() == __get_top_class_property_list().end())
674 {
675 m_class_layer.pop_back();
676 }
677 break;
678 }
679
680 case TypeTriviality::_NotTrivial:
681
682 if (__get_the_top_class_property_list_iterator() == __get_top_class_property_list().end()) // Pop the class layer if the iterator reached the end of the property list.
683 {
684 m_class_layer.pop_back();
685 break;
686 }
687
688 // Find the class/struct meta data that contains its memory layer.
689 auto l_search_result = m_property_registry.find(__get_metadata_of_the_property(__get_the_top_class_property_list_iterator()).m_typename);
690
691 // This is to serialize and deserialize containers and class instances that can be iterated through foreach.
692 FE::task_base* const l_foreach_task = framework_base::get_framework().get_method_reflection().retrieve(__get_serialization_task_name(__get_metadata_of_the_property(__get_the_top_class_property_list_iterator()).m_typename)); // Load method pointer.
693 if (l_foreach_task != nullptr) // is serializable with foreach?
694 {
695 FE::arguments<std::pmr::string&, const void*> l_task_args; // Any containers with begin() and end() can be serialized and deserialized.
696 l_task_args._first = out_ret_buffer_p;
697 l_task_args._second = reinterpret_cast<FE::byte*>(&object_p) + (l_offset_from_the_upmost_base_class_instance + __get_memory_offset_of_the_property(__get_the_top_class_property_list_iterator()));
698 (*l_foreach_task)(this, nullptr, &l_task_args); // The pointed task knows what to do with the arguments type casting.
699
700 // Move on to the next registered property of the class layer.
701 ++(__get_the_top_class_property_list_iterator());
702 }
703 else if (l_search_result != m_property_registry.end()) // push the meta data onto the stack if found.
704 {
705 l_offset_from_the_upmost_base_class_instance = __get_memory_offset_of_the_property(__get_the_top_class_property_list_iterator());
706
707 // Look for the next registered property of the class layer.
708 ++(__get_the_top_class_property_list_iterator());
709
710 // Push the member variable iterator and the class meta data to the class layer if the class meta data is found.
711 m_class_layer.emplace_back(&(l_search_result->second), l_search_result->second.begin());
712 __push_parent_class_layers_by_typename_string_recursive(l_search_result->first);
713 }
714
715 // if it reached the end.
716 if (__get_the_top_class_property_list_iterator() == __get_top_class_property_list().end())
717 {
718 m_class_layer.pop_back();
719 }
720 break;
721 }
722 }

725 template<typename T>

727 {
728 var::ptrdiff l_offset_from_the_upmost_base_class_instance = 0;
729 while (m_class_layer.empty() == false)
730 {
731 switch (__get_metadata_of_the_property(__get_the_top_class_property_list_iterator())._is_trivial)

737

739 FE::memcpy(reinterpret_cast<var::byte*>(&out_object_p) + (l_offset_from_the_upmost_base_class_instance + __get_memory_offset_of_the_property(__get_the_top_class_property_list_iterator())), FE::iterator_cast<FE::ASCII*>(m_position), __get_metadata_of_the_property(__get_the_top_class_property_list_iterator())._size_in_byte);

741
742 // Look for the next registered property of the class layer.
743 ++(__get_the_top_class_property_list_iterator());
744
745 // Pop the class layer if the iterator reached the end of the property list.

747 {
748 m_class_layer.pop_back();
749 }

766 if (l_foreach_task != nullptr) // is deserializable with foreach?
767 {
768 FE::arguments<void*> l_pointer_to_container; // Any containers with begin() and end() can be serialized and deserialized.
769 l_pointer_to_container._first = reinterpret_cast<var::byte*>(&out_object_p) + (l_offset_from_the_upmost_base_class_instance + __get_memory_offset_of_the_property(__get_the_top_class_property_list_iterator()));
770 (*l_foreach_task)(this, nullptr, &l_pointer_to_container); // The pointed task object_base knows what to do with the arguments type casting.
771
772 // Look for the next registered property of the class layer.
773 ++(__get_the_top_class_property_list_iterator());
774 }
775 else if (l_search_result != m_property_registry.end()) // push the meta data onto the stack if found.
776 {

780 ++(__get_the_top_class_property_list_iterator());

783 m_class_layer.emplace_back(&(l_search_result->second), l_search_result->second.begin());
784 __push_parent_class_layers_by_typename_string_recursive(l_search_result->first);
785 }
786
787 // if it reached the end.
788 if (__get_the_top_class_property_list_iterator() == __get_top_class_property_list().end())
789 {
790 m_class_layer.pop_back();
791 }
792 break;
793 }

797 template<class Container>
798 void __serialize_by_foreach_mutually_recursive(std::pmr::string& out_ret_buffer_p, const void* const data_p) noexcept
799 {
800 //FE_NEGATIVE_STATIC_ASSERT(FE::is_trivial<Container>::value == true, "Incorrect template argument type: serializable containers are not trivially constructible and destructible.");
801 static_assert(FE::is_serializable<Container>::value, "The container is unable to be serialized: the container type is not supported and not compatible to this system.");
802 FE_NEGATIVE_ASSERT(data_p == nullptr, "Aborting the serialization process: the pointer to the container is nullptr.");
803 const Container* const l_container = static_cast<const Container* const>(data_p);
804
805 if constexpr (FE::is_trivial<Container>::value == false)
806 {
807 if constexpr (FE::is_trivial<typename Container::value_type>::value == true)
808 {
809 if constexpr (std::is_array<Container>::value == true)
810 {
811 out_ret_buffer_p.append(reinterpret_cast<const char*>(l_container), sizeof(Container));
812 }
813 else
814 {
815 m_scalable_container_size_record.push_back(l_container->size());
816 out_ret_buffer_p.append(reinterpret_cast<const char*>(l_container->data()), sizeof(typename Container::value_type) * l_container->size());
817 }
818 }
819 else
820 {
821 m_scalable_container_size_record.push_back(l_container->size());
822 for (auto& element : *l_container)
823 {
824 if constexpr ((FE::is_serializable<typename Container::value_type>::value == true) && (FE::is_trivial<typename Container::value_type>::value == false))
825 {
826 __serialize_by_foreach_mutually_recursive<typename Container::value_type>(out_ret_buffer_p, &element);
827 }
828 else
829 {
830 __serialize_mutually_recursive<typename Container::value_type>(out_ret_buffer_p, element);
831 }
832 }
833 }
834 }

839 template<class Container>
840 void __deserialize_by_foreach_mutually_recursive(void* const data_p) noexcept
841 {
842 //FE_NEGATIVE_STATIC_ASSERT(FE::is_trivial<Container>::value == true, "Incorrect template argument type: serializable containers are not trivially constructible and destructible.");
843 static_assert(FE::is_serializable<Container>::value, "The container is unable to be deserialized: the container type is not supported and not compatible to this system.");
844 FE_NEGATIVE_ASSERT(data_p == nullptr, "Aborting the deserialization process: the pointer to the container is nullptr.");
845 Container* const l_container = static_cast<Container* const>(data_p);
846
847 if constexpr (FE::is_trivial<Container>::value == false)
848 {
849 l_container->resize(m_scalable_container_size_record.front());
850 m_scalable_container_size_record.pop_front();
851
852 if constexpr (FE::is_trivial<typename Container::value_type>::value == true)
853 {
854 if constexpr (std::is_array<Container>::value == true)
855 {
856 FE::memcpy(reinterpret_cast<var::byte*>(l_container), FE::iterator_cast<FE::ASCII*>(m_position), sizeof(Container));
857 m_position += sizeof(Container); // Iterate to the next bit.
858 }
859 else
860 {
861 FE::memcpy(reinterpret_cast<var::byte*>(l_container->data()), FE::iterator_cast<FE::ASCII*>(m_position), sizeof(typename Container::value_type) * l_container->size());
862 m_position += sizeof(typename Container::value_type) * l_container->size(); // Iterate to the next bit.
863 }
864 }
865 else
866 {
867 for (auto& element : *l_container)
868 {
869 if constexpr ((FE::is_serializable<typename Container::value_type>::value == true) && (FE::is_trivial<typename Container::value_type>::value == false))
870 {
871 __deserialize_by_foreach_mutually_recursive<typename Container::value_type>(&element);
872 }
873 else
874 {
875 __deserialize_mutually_recursive<typename Container::value_type>(element);
876 }
877 }
878 }
879 }
