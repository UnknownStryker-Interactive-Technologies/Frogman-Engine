905 {

910 if (l_prev != nullptr)

912 l_prev->m_next = l_next;
913 }

920 if (l_next != nullptr)

932 m_allocator.deallocate(FE::iterator_cast<node*>(pos_p), 1);

937 return end(); // return end() if the last element was erased.
938 }
939 return FE::iterator_cast<iterator>(l_next); // return the next element as the new position.
