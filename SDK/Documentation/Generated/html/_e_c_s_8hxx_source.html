366 template <class Archetype, typename ...Arguments>
367 FE::entity<Archetype> instanciate_entity(Arguments&& ...arguments_p) noexcept
368 {
369 static_assert(std::is_base_of_v<FE::archetype_base, Archetype>, "Static assertion failed: the template argument Archetype must be derived from FE::archetype_base.");
370 std::lock_guard<FE::mutex> l_lock(m_fiber_lock);
371
372 FE::archetype l_alloc_result = FE::make_owner<Archetype>( &m_archetype_pool, *this, std::forward<Arguments>(arguments_p)... );
373 l_alloc_result->m_component_view_table = typename FE::archetype_base::component_view_table(&m_memory_resource);
374
375 for (typename archetype_table::iterator entity_group = m_entity_list.begin(); entity_group != m_entity_list.end(); ++entity_group)
376 {
377 if (entity_group->get_size() == entity_group->max_entities)
378 {
379 continue;
380 }
381
382 FE::entity<Archetype> l_view = FE::downcast_owner_to_observer<Archetype>(l_alloc_result);
383 FE::size l_idx = entity_group->add_entity( std::move(l_alloc_result) );
384 l_view->m_group = entity_group;
385 l_view->m_index = l_idx;
386 return l_view;
387 }
388
389 if (l_alloc_result != nullptr) // is std::move()-ed? l_alloc_result must be nullptr if std::move()-ed.
390 { // could not find a group with free space; allocate a new group.
391 m_entity_list.emplace_front();
392 m_gc_root._entity_roots.push_back(m_entity_list.begin()); // add to the GC tracking list.
393
394 FE::entity<Archetype> l_view = FE::downcast_owner_to_observer<Archetype>(l_alloc_result);
395 FE::size l_idx = m_entity_list.begin()->add_entity(std::move(l_alloc_result));
396 l_view->m_group = m_entity_list.begin();
397 l_view->m_index = l_idx;
398 return l_view;
399 }
400
401 return entity<Archetype>();
402 }

404 template <class Archetype>
405 FE::entity<FE::archetype_base> instanciate_entity_from_initializer(const FE::framework::initializer& serialized_entity_p) noexcept
406 {
407 static_assert(std::is_base_of_v<FE::archetype_base, Archetype>, "Static assertion failed: the template argument Archetype must be derived from FE::archetype_base.");
408 FE::entity<FE::archetype_base> l_entity = ECS::instanciate_entity<Archetype>();
409
410 {
411 std::lock_guard<FE::mutex> l_lock(m_fiber_lock);
412
413 if (l_entity.is_valid() == false)
414 {
415 return entity<FE::archetype_base>();
416 }
417
418 for (const auto& [component_identifier, serialized_component] : serialized_entity_p)
419 {
420 constexpr FE::ASCII* l_class = "class";
421 constexpr FE::ASCII* l_struct = "struct";
422
423 m_buffer = component_identifier;
424 std::pmr::string::size_type l_pos = m_buffer.find(l_class);
425 if (l_pos == std::pmr::string::npos)
426 {
427 l_pos = m_buffer.find(l_struct);
428 FE_ASSERT(l_pos != std::pmr::string::npos, "Assertion failed: the component type name must start with 'class' or 'struct'.");
429 l_pos += std::strlen(l_struct);
430 }
431 else
432 {
433 l_pos += std::strlen(l_class);
434 }
435 m_buffer.erase(0, l_pos);
436
437 for (l_pos = m_buffer.find(' '); l_pos != std::pmr::string::npos; l_pos = m_buffer.find(' '))
438 {
439 m_buffer.erase(m_buffer.begin() + l_pos);
440 }
441 m_buffer.insert(0, "::");
442
443 FE::task_base* l_component_adder = FE::framework::framework_base::get_framework().get_method_reflection().retrieve(m_buffer);
444 FE_ASSERT(l_component_adder != nullptr, "Assertion failed: the component adder function is not found. The component type may not be registered.");
445
446 FE::component_view<FE::component_base> l_handle;
447 FE::arguments<FE::entity<FE::archetype_base>> l_arguments;
448 l_arguments._first = l_entity;
449 (*l_component_adder)(this, &l_handle, &l_arguments); // Perform reflection magic to add the component.
450 }
451 }
452
453 deserialize_entity(serialized_entity_p, l_entity);
454 return l_entity;
455 }

493 template <class Component, typename ...Arguments>
494 FE::component_view<Component> add_component(FE::entity<archetype_base> entt_p, Arguments&& ...arguments_p) noexcept
495 {
496 static_assert(std::is_base_of_v<FE::component_base, Component>, "Static assertion failed: the template argument Component must be derived from FE::component_base.");
497 FE_ASSERT(entt_p.is_valid() == true, "Assertion failed: the entity is not valid.");
498 std::lock_guard<FE::mutex> l_lock(m_fiber_lock);
499
500 typename component_table::iterator l_probe_result = m_component_table.find(FE::framework::reflection::type_id<Component>().hash_code());
501 if (l_probe_result == m_component_table.end())
502 {
503 typename component_table::mapped_type& l_list_and_allocator = m_component_table[FE::framework::reflection::type_id<Component>().hash_code()];
504 l_list_and_allocator._second.emplace_front(); // allocate a component pool.
505
506 m_gc_root._component_roots.push_back(l_list_and_allocator._second.begin()); // add to the GC tracking list.
507
508 l_probe_result = m_component_table.find(FE::framework::reflection::type_id<Component>().hash_code());
509 FE_ASSERT(l_probe_result != m_component_table.end(), "Assertion failed: failed to emplace a new list of components in the component table.");
510 }
511
512
513 FE::component l_alloc_result;
514 for (typename component_table::mapped_type::second_type::iterator component_group = l_probe_result->second._second.begin(); component_group != l_probe_result->second._second.end(); ++component_group)
515 {
516 if (component_group->get_size() == component_group->max_components) // true if the component pool is full.
517 {
518 continue;
519 }
520
521 l_alloc_result = FE::make_owner<Component>(&(l_probe_result->second._first), std::forward<Arguments>(arguments_p)...);
522 l_alloc_result->m_metadata = FE::make_owner<FE::internal::ECS::component_metadata>( &m_memory_resource );
523 l_alloc_result->m_metadata->_type_hash = FE::framework::reflection::type_id<Component>().hash_code();
524 l_alloc_result->m_metadata->_typename = FE::framework::reflection::type_id<Component>().name();
525 l_alloc_result->m_metadata->_gc_metadata = FE::make_owner<FE::internal::ECS::gc_metadata>(&m_memory_resource);
526 l_alloc_result->m_metadata->_gc_metadata->_member_components = typename FE::internal::ECS::gc_metadata::member_component_list_type(&m_memory_resource);
527
528 FE::component_view<Component> l_view = FE::downcast_owner_to_observer<Component>(l_alloc_result);
529
530 auto l_result = entt_p->m_component_view_table.emplace(FE::framework::reflection::type_id<Component>().hash_code(), l_view);
531 if (l_result.second == false)
532 {
533 FE_LOG(FE::log::Severity::_Warning, "FE ECS: add_component<T>() failed due to a pre-existing duplicate component.");
534 return component_view<Component>();
535 }
536
537 FE::size l_idx = component_group->add_component(std::move(l_alloc_result));
538 l_view->m_metadata->_group = component_group;
539 l_view->m_metadata->_index = l_idx;
540 return l_view;
541 }
542
543
544 // could not find a component pool with free space, allocate a new component pool!
545 if (l_alloc_result == nullptr) _FE_LIKELY_
546 {
547 // All components lists are full. Create a new one.
548 l_probe_result->second._second.emplace_front();
549 m_gc_root._component_roots.push_back(l_probe_result->second._second.begin()); // add to the GC tracking list.
550
551 l_alloc_result = FE::make_owner<Component>(&(l_probe_result->second._first), std::forward<Arguments>(arguments_p)...);
552 l_alloc_result->m_metadata = FE::make_owner<FE::internal::ECS::component_metadata>( &m_memory_resource );
553 l_alloc_result->m_metadata->_type_hash = FE::framework::reflection::type_id<Component>().hash_code();
554 l_alloc_result->m_metadata->_typename = FE::framework::reflection::type_id<Component>().name();
555 l_alloc_result->m_metadata->_gc_metadata = FE::make_owner<FE::internal::ECS::gc_metadata>(&m_memory_resource);
556 l_alloc_result->m_metadata->_gc_metadata->_member_components = typename FE::internal::ECS::gc_metadata::member_component_list_type(&m_memory_resource);
557
558 FE::component_view<Component> l_view = FE::downcast_owner_to_observer<Component>(l_alloc_result);
559
560 auto l_result = entt_p->m_component_view_table.emplace(FE::framework::reflection::type_id<Component>().hash_code(), l_view);
561 if (l_result.second == false)
562 {
563 FE_LOG(FE::log::Severity::_Warning, "FE ECS: add_component<T>() failed due to a pre-existing duplicate component.");
564 return component_view<Component>();
565 }
566
567 FE::size l_idx = l_probe_result->second._second.front().add_component(std::move(l_alloc_result));
568 l_view->m_metadata->_group = l_probe_result->second._second.begin();
569 l_view->m_metadata->_index = l_idx;
570 return l_view;
571 }
572
573 return component_view<Component>();
574 }
