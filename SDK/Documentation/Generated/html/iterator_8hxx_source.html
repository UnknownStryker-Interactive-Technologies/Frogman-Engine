618/*
619The contiguous_iterator class template in the FE namespace provides an iterator for traversing contiguous memory
620supporting various operations such as dereferencing, pointer arithmetic, and comparison, while ensuring safety against null pointers.
621*/
622template <typename T>
623class contiguous_iterator
624{
625public:
626 using iterator_category = std::random_access_iterator_tag;
627 using value_type = T;
628 using difference_type = var::ptrdiff;
629 using pointer = T*;
630 using reference = T&;
631 using const_pointer = const T*;
632 using const_reference = const T&;
633
634protected:
635 pointer m_iterator;
636
637public:
638 _FE_CONSTEXPR17_ contiguous_iterator() noexcept : m_iterator() {}
639 _FE_CONSTEXPR17_ contiguous_iterator(const_pointer const value_p) noexcept : m_iterator(const_cast<pointer>(value_p)) {}
640 _FE_CONSTEXPR17_ contiguous_iterator(const contiguous_iterator& other_p) noexcept : m_iterator(other_p.m_iterator) {}
641 _FE_CONSTEXPR17_ contiguous_iterator(contiguous_iterator&& other_p) noexcept : m_iterator(other_p.m_iterator) { other_p.m_iterator = nullptr; }
642 constexpr ~contiguous_iterator() noexcept {}
643
644 _FE_FORCE_INLINE_ constexpr boolean is_null() const noexcept { return m_iterator == nullptr; }
645
646 _FE_FORCE_INLINE_ constexpr reference operator*() noexcept
647 {
648 FE_NEGATIVE_ASSERT(m_iterator == nullptr, "${%s@0}: Unable to dereference a null iterator.", TO_STRING(FE::ErrorCode::_FatalMemoryError_1XX_NullPtr));

650 }

652 {

803struct is_frogman_iterator
804{
805 _FE_MAYBE_UNUSED_ static constexpr inline bool value = (
806 (FE::is_iterator<T>::value == true) ||
807 (FE::is_const_iterator<T>::value == true) ||
808 (FE::is_reverse_iterator<T>::value == true) ||

814template <typename T, typename = void>
815struct is_frogman_iterator_wrapper : std::false_type {};
816
817template <typename T>
818struct is_frogman_iterator_wrapper<T, std::void_t<typename T::wrapped_iterator_type>> : std::true_type {};
819
820/* Possible conversions
821* 1. Pointer type -> Frogman Engine Iterator type
822* 2. Frogman Engine Iterator type -> Pointer type
823* 3. Pointer type -> Pointer type
824* 4. Frogman Engine Iterator type -> Frogman Engine Iterator type
825* 5. STL Iterator type -> Pointer type (The opposite direction is not allowed).
826* 6. STL Iterator type -> Frogman Engine Iterator type (The opposite direction is not allowed).
827
828*/
829template<class To, class From>
830_FE_FORCE_INLINE_ constexpr To iterator_cast(From ptr_p) noexcept
831{
832 static_assert(FE::has_iterator_category<From>::value || std::is_pointer<From>::value, "Static assertion failure: template arguments must be a pointer type or an iterator type.");
833 static_assert(FE::has_iterator_category<To>::value || std::is_pointer<To>::value, "Static assertion failure: template arguments must be a pointer type or an iterator type.");
834
835 if constexpr (FE::is_frogman_iterator<From>::value == true)
836 {
837 if constexpr (std::is_pointer_v<To> == true)
838 {
839 return const_cast<To>(ptr_p.operator->());
840 }
841 else if constexpr (FE::is_frogman_iterator_wrapper<To>::value == true)
842 {
843 return To{ ptr_p };
844 }
845 else if constexpr (std::is_class_v<To> == true)
846 {
847 return To{ const_cast<typename To::pointer>(ptr_p.operator->()) };
848 }
849 }
850 else if constexpr (FE::is_frogman_iterator_wrapper<From>::value == true)
851 {
852 if constexpr (std::is_pointer_v<To> == true)
853 {
854 return const_cast<To>( ptr_p.unwrap().operator->());
855 }
856 else if constexpr (FE::is_frogman_iterator_wrapper<To>::value == true)
857 {
858 return To{ ptr_p.unwrap() };
859 }
860 else if constexpr (std::is_class_v<To> == true)
861 {
862 return To{ const_cast<typename To::pointer>( ptr_p.operator->() ) };
863 }
864 }
865 else if constexpr (std::is_class_v<From> == true)
866 {
867 if constexpr (std::is_pointer_v<To> == true)
868 {
869 return std::pointer_traits<To>::pointer_to(const_cast<typename From::reference>(*ptr_p));
870 }
871 else if constexpr (std::is_class_v<To> == true)
872 {
873 return To{ std::pointer_traits<typename From::pointer>::pointer_to(const_cast<typename From::reference>(*ptr_p)) };
874 }
875 }
876 else if constexpr (std::is_pointer_v<From> == true)
877 {
878 if constexpr (std::is_pointer_v<To> == true)
879 {
880 return (To)(ptr_p);
881 }
882 else if constexpr (FE::is_frogman_iterator_wrapper<To>::value == true)
883 {
884 return To{ typename To::wrapped_iterator_type{ const_cast<typename std::remove_const_t<From>>(ptr_p) } };
885 }
886 else if constexpr (std::is_class_v<To> == true)
887 {
888 return To{ const_cast<typename std::remove_const_t<From>>(ptr_p) };
889 }
890 }
891}
