        VERSION 0.0
        DESCRIPTION "Frogman Engine Framework library"
        LANGUAGES CXX)

${FE_FRAMEWORK_HEADERS}
${FE_FRAMEWORK_REFLECTION}
${FE_FRAMEWORK_SOURCES}
)

        SET_TARGET_PROPERTIES(${CMAKE_PROJECT_NAME} PROPERTIES
        ARCHIVE_OUTPUT_DIRECTORY_DEBUG          ${FE_FRAMEWORK_CMAKE_CURRENT_LIST_DIR}/../Binaries/X86-64/Windows/AVX512F/Debug

        ARCHIVE_OUTPUT_DIRECTORY_MINSIZEREL     ${FE_FRAMEWORK_CMAKE_CURRENT_LIST_DIR}/../Binaries/X86-64/Windows/AVX512F/MinSizeRel
        )
    ELSE()
        SET_TARGET_PROPERTIES(${CMAKE_PROJECT_NAME} PROPERTIES
        ARCHIVE_OUTPUT_DIRECTORY_DEBUG          ${FE_FRAMEWORK_CMAKE_CURRENT_LIST_DIR}/../Binaries/X86-64/Windows/AVX-SSE2/Debug
        ARCHIVE_OUTPUT_DIRECTORY_RELWITHDEBINFO ${FE_FRAMEWORK_CMAKE_CURRENT_LIST_DIR}/../Binaries/X86-64/Windows/AVX-SSE2/RelWithDebInfo
        ARCHIVE_OUTPUT_DIRECTORY_RELEASE        ${FE_FRAMEWORK_CMAKE_CURRENT_LIST_DIR}/../Binaries/X86-64/Windows/AVX-SSE2/Release
        ARCHIVE_OUTPUT_DIRECTORY_MINSIZEREL     ${FE_FRAMEWORK_CMAKE_CURRENT_LIST_DIR}/../Binaries/X86-64/Windows/AVX-SSE2/MinSizeRel
        )
    ENDIF()

ELSEIF(CMAKE_SYSTEM_NAME STREQUAL "Windows" AND TARGET_CPU_ARCHITECTURE STREQUAL "arm64")

ARCHIVE_OUTPUT_DIRECTORY_DEBUG          ${FE_FRAMEWORK_CMAKE_CURRENT_LIST_DIR}/../Binaries/ARM64/Windows/Debug
ARCHIVE_OUTPUT_DIRECTORY_RELWITHDEBINFO ${FE_FRAMEWORK_CMAKE_CURRENT_LIST_DIR}/../Binaries/ARM64/Windows/RelWithDebInfo
ARCHIVE_OUTPUT_DIRECTORY_RELEASE        ${FE_FRAMEWORK_CMAKE_CURRENT_LIST_DIR}/../Binaries/ARM64/Windows/Release

ARCHIVE_OUTPUT_DIRECTORY_DEBUG          ${FE_FRAMEWORK_CMAKE_CURRENT_LIST_DIR}/../Binaries/X86-64/Linux/Debug
ARCHIVE_OUTPUT_DIRECTORY_RELWITHDEBINFO ${FE_FRAMEWORK_CMAKE_CURRENT_LIST_DIR}/../Binaries/X86-64/Linux/RelWithDebInfo
